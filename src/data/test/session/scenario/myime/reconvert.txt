# Test for MS-IME keymap

# Test Henkan key (Reconvert) in Direct input mode.

SET_SELECTION_TEXT	集合
SEND_KEY	HENKAN
EXPECT_PREEDIT	集合
EXPECT_IN_ALL_CANDIDATE_WORDS	集合

RESET_CONTEXT

# Test Henkan key (Reconvert) in Precomposition mode.
SWITCH_COMPOSITION_MODE	HIRAGANA

SEND_KEY	HENKAN
EXPECT_PREEDIT	集合
EXPECT_IN_ALL_CANDIDATE_WORDS	集合

RESET_CONTEXT

# Reconvert Hiragana committed without conversion to test alternative spellings.
SWITCH_COMPOSITION_MODE	HIRAGANA

SEND_KEY	HENKAN
EXPECT_PREEDIT	集合
